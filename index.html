async function sendMessage() { const userInput = document.getElementById('userInput'); const text = userInput.value.trim(); if (!text && !pendingImage) return; const session = getCurrentSession(); // تحديث عنوان المحادثة أول ما يبدأ المستخدم بالكتابة if (session.messages.length === 0 && text) { session.title = text.length > 20 ? text.substring(0, 20) + "..." : text; renderSessionsList(); } else if (session.messages.length === 0 && pendingImage) { session.title = "صورة مرفقة"; renderSessionsList(); } const currentImg = pendingImage; pendingImage = null; const imagePreviewName = document.getElementById('image-preview-name'); if (imagePreviewName) imagePreviewName.textContent = ""; if (currentImg) { session.messages.push({ role: "user", content: text || "تفضل تحليل هذه الصورة", type: "user-image", imageUrl: currentImg }); } else { session.messages.push({ role: "user", content: text }); } userInput.value = ''; renderChatBox(); const sendBtn = document.getElementById('sendBtn'); if (sendBtn) sendBtn.disabled = true; const chatBox = document.getElementById('chatBox'); const typingContainer = document.createElement('div'); typingContainer.className = 'message-container ai'; typingContainer.id = 'typingIndicator'; typingContainer.innerHTML = `
جاري التفكير... ⏳
`; chatBox.appendChild(typingContainer); chatBox.scrollTop = chatBox.scrollHeight; const lowerText = text.toLowerCase(); const isImageRequest = !currentImg && (lowerText.startsWith("ارسم") || lowerText.startsWith("تخيل") || lowerText.includes("صورة لـ") || lowerText.includes("صورة") || lowerText.includes("draw") || lowerText.includes("image of")); if (isImageRequest) { let promptText = text.replace(/^(ارسم|تخيل|draw|image of)\s*/i, "").trim(); if (!promptText) promptText = text; const imageUrl = `https://pollinations.ai/p/${encodeURIComponent(promptText)}?key=${POLLINATIONS_API_KEY}`; setTimeout(() => { if (document.getElementById('typingIndicator')) document.getElementById('typingIndicator').remove(); if (sendBtn) sendBtn.disabled = false; const replyText = `سم يا وليد، هذي هي الصورة بناءً على طلبك "${promptText}":`; session.messages.push({ role: "assistant", content: replyText, type: "image", imageUrl: imageUrl }); saveData(); renderChatBox(); }, 1200); } else { try { let systemPrompt = "تحدث باللهجة السعودية أو الفصحى، أنت مساعد ذكي محترم جداً باسم Saudi AI. المبسطة بكل أدب واحترام وابتعد عن الألفاظ البذيئة تماماً"; // تجهيز الرسائل بشكل متوافق تماماً مع Groq Vision API let apiMessages = [{ role: "system", content: systemPrompt }]; // نأخذ آخر صورة تم إرسالها إن وجدت في المحادثة let lastImage = currentImg; if (!lastImage) { for (let i = session.messages.length - 1; i >= 0; i--) { if (session.messages[i].type === "user-image" && session.messages[i].imageUrl) { lastImage = session.messages[i].imageUrl; break; } } } if (lastImage) { // إذا فيه صورة، ندمج النص والصورة بالشكل الصحيح لـ Groq Vision let chatHistoryText = ""; session.messages.forEach(m => { if (m.role === "user") chatHistoryText += `المستخدم: ${m.content}\n`; if (m.role === "assistant") chatHistoryText += `المساعد: ${m.content}\n`; }); apiMessages.push({ role: "user", content: [ { type: "text", text: chatHistoryText + "\n(اشرح لي وش تشوف بهذه الصورة بالتفصيل)" }, { type: "image_url", image_url: { url: lastImage } } ] }); } else { // نص عادي بدون صورة session.messages.forEach(m => { if (m.role === 'user' || m.role === 'assistant') { apiMessages.push({ role: m.role, content: m.content }); } }); } const response = await fetch("https://api.groq.com/openai/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${GROQ_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: lastImage ? "llama-3.2-11b-vision-preview" : "llama-3.3-70b-versatile", messages: apiMessages, max_tokens: 1024 }) }); const data = await response.json(); if (document.getElementById('typingIndicator')) document.getElementById('typingIndicator').remove(); if (sendBtn) sendBtn.disabled = false; if (data.choices && data.choices.length > 0) { const reply = data.choices[0].message.content; session.messages.push({ role: "assistant", content: reply }); saveData(); renderChatBox(); } else { throw new Error(data.error?.message || "خطأ في استجابة الخادم."); } } catch (error) { console.error("Fetch Error:", error); if (document.getElementById('typingIndicator')) document.getElementById('typingIndicator').remove(); if (sendBtn) sendBtn.disabled = false; alert("حدث خطأ في الاتصال بالشبكة أو الخادم."); } } }
